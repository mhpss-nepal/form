<script>
(function () {
  "use strict";
  var C = window.CODES, L = window.L1, KIND = "referral";
  var $ = function (id) { return document.getElementById(id); };

  L.fillSelect($("org"), C.ORGS.concat([{ code: "OTHER", name: "Other — not listed", np: "अन्य — सूचीमा नभएको", np_src: "draft" }]));
  ["org", "direction"].forEach(function (id) {
    var v = L.recall(id); if (v) $(id).value = v;
    $(id).addEventListener("change", function () { L.remember(id, $(id).value); });
  });

  function record() {
    return {
      schema: "mhpss-np-referral-outcome/0.1.0-draft",
      kind: "referral",
      record_type: "outcome",
      org: $("org").value,
      direction: $("direction").value,
      contact_code: $("ccode").value.trim().toUpperCase(),
      date_ad: $("dateAD").value,
      arrived: $("arrived").value,
      arrival_confirmed_date: $("arrDate").value,
      confirmed_by: $("confBy").value,
      barrier: $("barrier").value
    };
  }
  var COLS = Object.keys(record()).concat(["_saved"]);
  function payload() { $("payload").textContent = JSON.stringify(record(), null, 2); }

  function problems(r) {
    var e = [];
    if (!r.org) e.push("Choose your organisation.");
    if (!r.direction) e.push("Say whether the referral went out from you or came in to you.");
    if (!r.contact_code) e.push("Give the contact code the referral was made under.");
    if (r.contact_code && !/^NP-[0-9A-Z]{5}-[0-9A-Z]{5}$/.test(r.contact_code))
      e.push("The contact code should look like NP-ABCDE-12345.");
    if (!r.date_ad) e.push("Give the date of the referral.");
    if (r.date_ad && r.date_ad > L.today()) e.push("The referral date is in the future.");
    if (!r.arrived) e.push("Say what happened. If it is not yet known, there is nothing to file yet.");
    if (r.arrived === "Y" && !r.arrival_confirmed_date)
      e.push("If arrival is confirmed, give the date it was confirmed.");
    if (r.arrival_confirmed_date && r.date_ad && r.arrival_confirmed_date < r.date_ad)
      e.push("Arrival cannot be confirmed before the referral was made.");
    if (r.arrived === "N" && !r.barrier)
      e.push("If the person did not arrive, say what got in the way — that is the point of recording it.");
    return e;
  }

  function showCount() {
    var n = L.all(KIND).length;
    $("count").textContent = n ? n + " referral outcome(s) on this device" : "nothing saved on this device yet";
  }
  showCount();

  Array.prototype.forEach.call(document.querySelectorAll("input,select"), function (el) {
    el.addEventListener("change", payload);
    if (el.tagName === "INPUT" && el.type === "text") el.addEventListener("input", payload);
  });

  $("f").addEventListener("submit", function (ev) {
    ev.preventDefault();
    var r = record(), e = problems(r);
    if (e.length) {
      $("errs").innerHTML = '<div class="banner stop"><b>Not saved.</b><ul><li>' +
        e.map(function (x) { return x.replace(/</g, "&lt;"); }).join("</li><li>") + "</li></ul></div>";
      $("errs").scrollIntoView({ block: "center" });
      return;
    }
    var res = L.save(KIND, r);
    $("ccode").value = ""; $("arrived").value = ""; $("arrDate").value = "";
    $("confBy").value = ""; $("barrier").value = "";
    $("errs").innerHTML = '<div class="banner ok"><b>Saved on this device.</b> ' +
      (res.persisted ? "" : "Storage is blocked — export before closing. ") +
      "It is matched to the referral by the contact code and the date of referral.</div>";
    showCount(); payload();
  });

  $("csvBtn").addEventListener("click", function () {
    var list = L.all(KIND);
    if (!list.length) { alert("Nothing saved on this device yet."); return; }
    L.download("mhpss_np_referral_outcomes_" + L.stamp() + ".csv", L.toCSV(list, COLS), "text/csv");
  });
  $("clearBtn").addEventListener("click", function () {
    if (!L.all(KIND).length) { alert("Nothing saved on this device yet."); return; }
    if (confirm("Delete every referral outcome held in this browser? This cannot be undone.")) {
      L.clear(KIND); showCount(); $("errs").innerHTML = '<div class="banner"><b>Cleared.</b></div>';
    }
  });

  if (!L.storageWorks()) $("storeWarn").textContent = " This browser is blocking storage, so entries live only while this page is open.";
  payload();
})();
</script>
